cmake_minimum_required(VERSION 3.24)
project(VoidSentinel VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(voidsentinel_core STATIC src/monitor.cpp src/results.cpp src/filter.cpp src/session.cpp)
target_include_directories(voidsentinel_core PUBLIC src)
target_compile_definitions(voidsentinel_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(voidsentinel_core PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(voidsentinel_core PUBLIC advapi32 tdh)
add_executable(VoidSentinel WIN32 src/main.cpp src/rules_ui.cpp src/app.rc)
set_property(SOURCE src/app.rc PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/VoidSentinel.windows.ico;${CMAKE_CURRENT_SOURCE_DIR}/src/resource.h")
target_compile_options(VoidSentinel PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_options(VoidSentinel PRIVATE /MANIFEST:NO)
target_link_libraries(VoidSentinel PRIVATE voidsentinel_core comctl32 comdlg32 shell32 ole32)
enable_testing()
foreach(test_name results filter)
    add_executable(${test_name}_test tests/${test_name}_test.cpp)
    target_compile_options(${test_name}_test PRIVATE /W4 /utf-8 /permissive- /EHsc)
    target_link_libraries(${test_name}_test PRIVATE voidsentinel_core)
    add_test(NAME ${test_name} COMMAND ${test_name}_test)
endforeach()

# Development-only UI build: same controls and logic, isolated settings, normal user privileges.
add_executable(VoidSentinelUiTest EXCLUDE_FROM_ALL WIN32 src/main.cpp src/rules_ui.cpp src/app.rc)
target_compile_definitions(VoidSentinelUiTest PRIVATE VOIDSENTINEL_UI_TEST)
target_compile_options(VoidSentinelUiTest PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_options(VoidSentinelUiTest PRIVATE /MANIFEST:NO)
target_link_libraries(VoidSentinelUiTest PRIVATE voidsentinel_core comctl32 comdlg32 shell32 ole32)
set_target_properties(VoidSentinelUiTest PROPERTIES OUTPUT_NAME VoidSentinel RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ui-test")
