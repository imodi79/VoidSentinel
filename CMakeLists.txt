cmake_minimum_required(VERSION 3.24)
project(VoidSentinel VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(VoidSentinel WIN32 src/main.cpp src/monitor.cpp src/results.cpp src/app.rc)
set_property(SOURCE src/app.rc PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/VoidSentinel.windows.ico;${CMAKE_CURRENT_SOURCE_DIR}/src/resource.h")
target_compile_definitions(VoidSentinel PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(VoidSentinel PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_options(VoidSentinel PRIVATE /MANIFEST:NO)
target_link_libraries(VoidSentinel PRIVATE comctl32 comdlg32 advapi32 tdh shell32 ole32)
enable_testing()
add_executable(results_test tests/results_test.cpp src/results.cpp src/monitor.cpp)
target_include_directories(results_test PRIVATE src)
target_compile_definitions(results_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(results_test PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(results_test PRIVATE advapi32 tdh)
add_test(NAME results COMMAND results_test)
